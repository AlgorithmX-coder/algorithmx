/* PLATFORM SUBSTRATE — dense machined gunmetal the reactor arcs are recessed
 * into. Concentric PANEL BANDS divided into plates by radial seams, each plate
 * tone-varied with a lit outer rim + dark inner shadow, sparse inset greebles
 * and tiny embedded light dots. NORMAL-blended so it paints real darks (the
 * manufactured detail in the reference). */
function makePlatformSubstrateTexture(): THREE.Texture | null {
  if (typeof document === "undefined") return null;
  const S = 2048;
  const c = document.createElement("canvas");
  c.width = c.height = S;
  const ctx = c.getContext("2d");
  if (!ctx) return null;
  ctx.imageSmoothingEnabled = true;
  ctx.imageSmoothingQuality = "high";
  ctx.clearRect(0, 0, S, S);
  const cx = S / 2;
  const cy = S / 2;
  const Rmax = S * 0.46;
  const hash = (n: number) => {
    const s = Math.sin(n * 127.1) * 43758.5453;
    return s - Math.floor(s);
  };

  /* dark gunmetal base, radial, fading to transparent before the edge */
  const base = ctx.createRadialGradient(cx, cy, 0, cx, cy, Rmax * 1.02);
  base.addColorStop(0, "rgba(20,26,36,0.96)");
  base.addColorStop(0.5, "rgba(13,18,26,0.9)");
  base.addColorStop(0.8, "rgba(7,10,16,0.6)");
  base.addColorStop(1, "rgba(0,0,0,0)");
  ctx.fillStyle = base;
  ctx.fillRect(0, 0, S, S);

  const ann = (r0: number, r1: number, a0: number, a1: number) => {
    ctx.beginPath();
    ctx.arc(cx, cy, r1, a0, a1);
    ctx.arc(cx, cy, r0, a1, a0, true);
    ctx.closePath();
  };

  /* concentric machined panel bands divided into plates by radial seams */
  const bandEdges = [0.06, 0.13, 0.2, 0.27, 0.34, 0.42, 0.5, 0.58, 0.66, 0.75, 0.84, 0.93, 1.0];
  for (let b = 0; b < bandEdges.length - 1; b++) {
    const r0 = bandEdges[b] * Rmax;
    const r1 = bandEdges[b + 1] * Rmax;
    const rMid = (r0 + r1) / 2;
    const fade = Math.max(0.12, 1 - rMid / Rmax);
    const panels = 10 + b * 3;
    for (let s = 0; s < panels; s++) {
      const a0 = (s / panels) * Math.PI * 2;
      const a1 = ((s + 1) / panels) * Math.PI * 2;
      const seamGap = 0.006 + hash(b * 7.3 + s) * 0.004;
      const pa0 = a0 + seamGap;
      const pa1 = a1 - seamGap;
      const tone = 14 + Math.floor(hash(b * 3.1 + s * 1.7) * 16);
      ann(r0 + 1.5, r1 - 1.5, pa0, pa1);
      ctx.fillStyle = `rgba(${tone},${tone + 5},${tone + 12},${0.55 * fade})`;
      ctx.fill();
      /* lit outer rim bevel */
      ctx.beginPath();
      ctx.arc(cx, cy, r1 - 1.5, pa0, pa1);
      ctx.strokeStyle = `rgba(92,118,148,${0.3 * fade})`;
      ctx.lineWidth = 1.4;
      ctx.stroke();
      /* dark inner shadow */
      ctx.beginPath();
      ctx.arc(cx, cy, r0 + 2, pa0, pa1);
      ctx.strokeStyle = `rgba(0,1,4,${0.5 * fade})`;
      ctx.lineWidth = 1.2;
      ctx.stroke();
      /* inset greeble on ~38% of plates */
      if (hash(b * 5.9 + s * 2.3) > 0.62) {
        const gr0 = r0 + (r1 - r0) * 0.32;
        const gr1 = r0 + (r1 - r0) * 0.7;
        const ga0 = pa0 + (pa1 - pa0) * 0.25;
        const ga1 = pa0 + (pa1 - pa0) * 0.75;
        ann(gr0, gr1, ga0, ga1);
        ctx.fillStyle = `rgba(4,7,12,${0.5 * fade})`;
        ctx.fill();
        ctx.strokeStyle = `rgba(72,96,122,${0.18 * fade})`;
        ctx.lineWidth = 0.8;
        ctx.stroke();
      }
      /* tiny embedded light dot on a few plates (mostly cyan, rare warm) */
      const lh = hash(b * 8.1 + s * 4.7);
      if (lh > 0.9) {
        const warm = lh > 0.965;
        const am = (pa0 + pa1) / 2;
        const lx = cx + Math.cos(am) * rMid;
        const ly = cy + Math.sin(am) * rMid;
        ctx.fillStyle = warm
          ? `rgba(255,168,78,${0.7 * fade})`
          : `rgba(120,210,255,${0.7 * fade})`;
        ctx.beginPath();
        ctx.arc(lx, ly, 1.7, 0, Math.PI * 2);
        ctx.fill();
      }
    }
  }

  /* near-camera fade */
  ctx.globalCompositeOperation = "destination-out";
  const ncf = ctx.createLinearGradient(0, 0, 0, S);
  ncf.addColorStop(0.0, "rgba(0,0,0,0)");
  ncf.addColorStop(0.52, "rgba(0,0,0,0)");
  ncf.addColorStop(0.66, "rgba(0,0,0,0.6)");
  ncf.addColorStop(0.8, "rgba(0,0,0,0.92)");
  ncf.addColorStop(1.0, "rgba(0,0,0,1)");
  ctx.fillStyle = ncf;
  ctx.fillRect(0, 0, S, S);
  ctx.globalCompositeOperation = "source-over";

  const tex = new THREE.CanvasTexture(c);
  tex.wrapS = THREE.ClampToEdgeWrapping;
  tex.wrapT = THREE.ClampToEdgeWrapping;
  tex.colorSpace = THREE.SRGBColorSpace;
  tuneGroundTexture(tex);
  tex.needsUpdate = true;
  return tex;
}

/* REACTOR ARCS (illumination) — concentric arc-SEGMENT rings glowing in a
 * disciplined cyan -> electric-blue -> sparse-violet family (NO white cores).
 * Brightest in a mid radius band, fading inner + outer, with dotted tech
 * segments on some arcs and a fine concentric hub. Additive; sits inside the
 * substrate grooves. The "soft" / "deep" variants are blurred ghosts for the
 * parallax strata. */
function makeInstrumentPlateTexture(
  variant: "surface" | "soft" | "deep" = "surface",
): THREE.Texture | null {
  if (typeof document === "undefined") return null;
  const S = 2048;
  const c = document.createElement("canvas");
  c.width = c.height = S;
  const ctx = c.getContext("2d");
  if (!ctx) return null;
  ctx.imageSmoothingEnabled = true;
  ctx.imageSmoothingQuality = "high";
  ctx.clearRect(0, 0, S, S);
  const soft = variant === "soft";
  const deep = variant === "deep";
  const cx = S / 2;
  const cy = S / 2;
  const Rmax = S * (deep ? 0.46 : soft ? 0.42 : 0.4);
  const hash = (n: number) => {
    const s = Math.sin(n * 127.1) * 43758.5453;
    return s - Math.floor(s);
  };
  /* brightest mid-band falloff, hard cut past the rim */
  const bandAlpha = (rN: number) => {
    const d = rN - 0.42;
    const g = Math.exp(-(d * d) / 0.1);
    const outer = rN > 0.9 ? Math.max(0, 1 - (rN - 0.9) / 0.1) : 1;
    return g * outer;
  };

  /* faint central energy pool under the chassis (kept) */
  if (!deep) {
    const pool = ctx.createRadialGradient(cx, cy, 0, cx, cy, S * 0.2);
    pool.addColorStop(0, soft ? "rgba(0,200,255,0.08)" : "rgba(0,210,255,0.22)");
    pool.addColorStop(0.5, soft ? "rgba(0,170,255,0.03)" : "rgba(0,170,255,0.08)");
    pool.addColorStop(1, "rgba(0,0,0,0)");
    ctx.fillStyle = pool;
    ctx.fillRect(0, 0, S, S);
  }

  ctx.lineCap = "round";
  const glow = (blur: number, col: string) => {
    ctx.shadowColor = col;
    ctx.shadowBlur = blur;
  };
  const noGlow = () => {
    ctx.shadowBlur = 0;
  };

  const rings = [0.14, 0.22, 0.3, 0.39, 0.48, 0.58, 0.68, 0.8, 0.92];
  for (let ri = 0; ri < rings.length; ri++) {
    const rN = rings[ri];
    const rPx = rN * Rmax;
    const a = bandAlpha(rN) * (soft ? 0.55 : deep ? 0.4 : 1.0);
    if (a < 0.02) continue;
    /* disciplined palette: mostly cyan, two violet accent rings, two blue */
    const isViolet = ri === 2 || ri === 6;
    const isBlue = ri === 4 || ri === 7;
    const rgb = isViolet ? "122,92,255" : isBlue ? "42,120,255" : "0,216,255";
    const lite = isViolet ? "180,160,255" : isBlue ? "150,200,255" : "150,238,255";
    const segs = ri < 2 ? 1 : 2 + (ri % 4);
    for (let s = 0; s < segs; s++) {
      let a0: number;
      let a1: number;
      if (segs === 1) {
        a0 = -Math.PI * 0.92;
        a1 = Math.PI * 0.92;
      } else {
        const span = (Math.PI * 2) / segs;
        const j = (hash(ri * 9.7 + s * 3.3) - 0.5) * span * 0.4;
        const gapFrac = 0.16 + hash(ri * 4.1 + s) * 0.2;
        a0 = s * span + j + span * gapFrac * 0.5;
        a1 = (s + 1) * span + j - span * gapFrac * 0.5;
      }
      /* soft glow underlay (bloom feeds on this) */
      ctx.strokeStyle = `rgba(${rgb},${a * (soft ? 0.5 : 0.4)})`;
      ctx.lineWidth = soft ? 5 : deep ? 3 : 3.4;
      glow(soft ? 9 : 6, `rgba(${rgb},0.9)`);
      ctx.beginPath();
      ctx.arc(cx, cy, rPx, a0, a1);
      ctx.stroke();
      noGlow();
      if (!soft && !deep) {
        /* crisp colour body */
        ctx.strokeStyle = `rgba(${rgb},${a})`;
        ctx.lineWidth = 1.8;
        ctx.beginPath();
        ctx.arc(cx, cy, rPx, a0, a1);
        ctx.stroke();
        /* lighter SAME-HUE inner core (never white) */
        ctx.strokeStyle = `rgba(${lite},${a * 0.8})`;
        ctx.lineWidth = 0.8;
        ctx.beginPath();
        ctx.arc(cx, cy, rPx, a0, a1);
        ctx.stroke();
        /* dotted tech segments on some arcs */
        if (hash(ri * 2.7 + s) > 0.5) {
          const n = 16;
          for (let k = 0; k < n; k++) {
            if (hash(ri * 11 + s * 3 + k) > 0.55) continue;
            const aa = a0 + (a1 - a0) * (k / (n - 1));
            ctx.fillStyle = `rgba(${lite},${a * 0.9})`;
            ctx.beginPath();
            ctx.arc(cx + Math.cos(aa) * rPx, cy + Math.sin(aa) * rPx, 1.5, 0, Math.PI * 2);
            ctx.fill();
          }
        }
      }
    }
  }

  /* fine concentric hub + soft centre */
  if (!soft && !deep) {
    for (const hr of [0.04, 0.06, 0.085]) {
      ctx.strokeStyle = "rgba(0,216,255,0.5)";
      ctx.lineWidth = 1.0;
      ctx.beginPath();
      ctx.arc(cx, cy, hr * Rmax, 0, Math.PI * 2);
      ctx.stroke();
    }
    const core = ctx.createRadialGradient(cx, cy, 0, cx, cy, S * 0.04);
    core.addColorStop(0, "rgba(130,232,255,0.45)");
    core.addColorStop(1, "rgba(130,232,255,0)");
    ctx.fillStyle = core;
    ctx.fillRect(0, 0, S, S);
  }

  /* near-camera fade */
  ctx.globalCompositeOperation = "destination-out";
  const ncf = ctx.createLinearGradient(0, 0, 0, S);
  ncf.addColorStop(0.0, "rgba(0,0,0,0)");
  ncf.addColorStop(0.52, "rgba(0,0,0,0)");
  ncf.addColorStop(0.66, "rgba(0,0,0,0.6)");
  ncf.addColorStop(0.8, "rgba(0,0,0,0.92)");
  ncf.addColorStop(1.0, "rgba(0,0,0,1)");
  ctx.fillStyle = ncf;
  ctx.fillRect(0, 0, S, S);
  ctx.globalCompositeOperation = "source-over";

  const tex = new THREE.CanvasTexture(c);
  tex.wrapS = THREE.ClampToEdgeWrapping;
  tex.wrapT = THREE.ClampToEdgeWrapping;
  tex.colorSpace = THREE.SRGBColorSpace;
  tuneGroundTexture(tex);
  tex.needsUpdate = true;
  return tex;
}
